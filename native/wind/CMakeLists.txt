cmake_minimum_required(VERSION 3.22)
project(alpine_wind LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT GODOT_CPP_PATH)
    message(FATAL_ERROR "Use scripts/build_wind.ps1 to supply pinned GODOT_CPP_PATH")
endif()
set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
add_subdirectory("${GODOT_CPP_PATH}" godot-cpp)
add_library(alpine_wind SHARED wind_stream.cpp sfx_stream.cpp)
target_link_libraries(alpine_wind PRIVATE godot-cpp)
target_compile_definitions(alpine_wind PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(alpine_wind PROPERTIES PREFIX "" OUTPUT_NAME "alpine_wind.windows.x86_64")
add_executable(wind_dsp_test wind_dsp_test.cpp)
add_executable(sfx_dsp_test sfx_dsp_test.cpp)
if(MSVC)
    target_compile_options(alpine_wind PRIVATE /O2 /W4)
    target_compile_options(wind_dsp_test PRIVATE /O2 /W4)
    target_compile_options(sfx_dsp_test PRIVATE /O2 /W4)
endif()
enable_testing()
add_test(NAME wind_dsp COMMAND wind_dsp_test)
add_test(NAME sfx_dsp COMMAND sfx_dsp_test)
