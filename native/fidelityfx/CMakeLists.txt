cmake_minimum_required(VERSION 3.22)
project(alpine_fidelityfx LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT FIDELITYFX_SDK)
    message(FATAL_ERROR "Provide the pinned FIDELITYFX_SDK directory")
endif()
set(FFX_ROOT "${FIDELITYFX_SDK}/Kits/FidelityFX")
add_executable(fidelityfx_probe probe.cpp)
target_compile_definitions(fidelityfx_probe PRIVATE _WINDOWS NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(fidelityfx_probe PRIVATE "${FFX_ROOT}/api/include" "${FFX_ROOT}/upscalers/include" "${FFX_ROOT}/framegeneration/include")
target_link_libraries(fidelityfx_probe PRIVATE d3d12 dxgi)
foreach(component loader upscaler framegeneration)
    add_custom_command(TARGET fidelityfx_probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FFX_ROOT}/signedbin/amd_fidelityfx_${component}_dx12.dll" "$<TARGET_FILE_DIR:fidelityfx_probe>")
endforeach()
