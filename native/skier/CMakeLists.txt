cmake_minimum_required(VERSION 3.22)
project(alpine_skier LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The current native skier build targets Windows x64/MSVC.")
endif()
if(NOT GODOT_CPP_PATH OR NOT GODOT_CPP_BUILD)
  message(FATAL_ERROR "Use scripts/build_skier.ps1 with the prepared wind godot-cpp dependency.")
endif()
add_library(alpine_skier SHARED anatomy.cpp body_fit.cpp register_types.cpp)
target_include_directories(alpine_skier PRIVATE
  "${GODOT_CPP_PATH}/include" "${GODOT_CPP_PATH}/gdextension"
  "${GODOT_CPP_BUILD}/godot-cpp/gen/include")
target_link_libraries(alpine_skier PRIVATE "${GODOT_CPP_BUILD}/bin/libgodot-cpp.windows.template_debug.x86_64.lib")
target_compile_definitions(alpine_skier PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN WINDOWS_ENABLED TYPED_METHOD_BIND GDEXTENSION DEBUG_ENABLED _HAS_EXCEPTIONS=0 THREADS_ENABLED)
target_compile_options(alpine_skier PRIVATE /O2 /W4 /fp:precise)
set_target_properties(alpine_skier PROPERTIES PREFIX "" OUTPUT_NAME "alpine_skier.windows.x86_64")
