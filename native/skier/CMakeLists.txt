cmake_minimum_required(VERSION 3.22)
project(alpine_skier LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(APPLE)
  # macOS arm64: build the pinned godot-cpp in-tree (scripts/build_skier_macos.sh)
  # and keep strict IEEE semantics so the kernel matches the GDScript reference
  # evaluated by the same engine build. No fused multiply-add contraction.
  if(NOT GODOT_CPP_PATH)
    message(FATAL_ERROR "Use scripts/build_skier_macos.sh to supply the pinned GODOT_CPP_PATH.")
  endif()
  set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "" FORCE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "" FORCE)
  set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "" FORCE)
  set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
  add_subdirectory("${GODOT_CPP_PATH}" godot-cpp)
  add_library(alpine_skier SHARED anatomy.cpp body_fit.cpp register_types.cpp)
  target_link_libraries(alpine_skier PRIVATE godot-cpp)
  target_compile_options(alpine_skier PRIVATE -O2 -Wall -ffp-contract=off -fno-fast-math)
  set_target_properties(alpine_skier PROPERTIES PREFIX "" OUTPUT_NAME "alpine_skier.macos.arm64" SUFFIX ".dylib")
else()
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The current native skier build targets Windows x64/MSVC or macOS arm64/clang.")
endif()
if(NOT GODOT_CPP_PATH OR NOT GODOT_CPP_BUILD)
  message(FATAL_ERROR "Use scripts/build_skier.ps1 with the prepared wind godot-cpp dependency.")
endif()
add_library(alpine_skier SHARED anatomy.cpp body_fit.cpp register_types.cpp)
target_include_directories(alpine_skier PRIVATE
  "${GODOT_CPP_PATH}/include" "${GODOT_CPP_PATH}/gdextension"
  "${GODOT_CPP_BUILD}/godot-cpp/gen/include")
target_link_libraries(alpine_skier PRIVATE "${GODOT_CPP_BUILD}/bin/libgodot-cpp.windows.template_debug.x86_64.lib")
target_compile_definitions(alpine_skier PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN WINDOWS_ENABLED TYPED_METHOD_BIND GDEXTENSION DEBUG_ENABLED _HAS_EXCEPTIONS=0 THREADS_ENABLED)
target_compile_options(alpine_skier PRIVATE /O2 /W4 /fp:precise)
set_target_properties(alpine_skier PROPERTIES PREFIX "" OUTPUT_NAME "alpine_skier.windows.x86_64")
endif()
